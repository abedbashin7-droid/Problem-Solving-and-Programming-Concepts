Start
    Declare Real mun1, mun2, mun3
    
    Display "Enter the first number:"
    Input mun1
    
    Display "Enter the second number:"
    Input mun2
    
    Display "Enter the third number:"
    Input mun3
    
    If mun1 <= mun2 Then
        If mun2 <= mun3 Then
            Display mun1, mun2, mun3
        Else
            If mun3 <= mun1 Then
                Display mun3, mun1, mun2
            Else
                Display mun1, mun3, mun2
            End If
        End If
    Else
        If mun1 < mun3 Then
            Display mun2, mun1, mun3
        Else
            If mun2 >= mun3 Then
                Display mun3, mun2, mun1
            Else
                Display mun2, mun3, mun1
            End If
        End If
    End If
End
